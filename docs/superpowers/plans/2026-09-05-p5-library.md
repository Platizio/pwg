```ts
/**
 * The two transforms every article body gets on the way in, done to the
 * string once rather than across thirty content files: every <h2> gets a
 * stable id so the contents rail can link to it, and every <table> gets the
 * scroll container the prose sheet styles. Pure, so the node suite can pin
 * them. No DOM, no React.
 */

export interface Section { id: string; label: string }

/** A heading's inner HTML as an anchor: lowercase, hyphenated, at most 64. */
export function slugifyHeading(html: string): string {
  return html
    .replace(/<[^>]+>/g, " ")
    .replace(/&amp;/g, " and ")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
    .slice(0, 64);
}

/** Rough word count from HTML, for the schema and the masthead. */
export function countWords(html: string): number {
  return html
    .replace(/<[^>]+>/g, " ")
    .replace(/&[a-z]+;/gi, " ")
    .split(/\s+/)
    .filter(Boolean).length;
}

export function prepareBody(html: string): { html: string; sections: Section[] } {
  const sections: Section[] = [];
  const taken = new Set<string>();

  const withIds = html.replace(/<h2>([\s\S]*?)<\/h2>/g, (_m, inner: string) => {
    const label = inner.replace(/<[^>]+>/g, "").replace(/&amp;/g, "&").trim();
    const base = slugifyHeading(inner) || "section";
    let id = base;
    let n = 2;
    while (taken.has(id)) id = `${base}-${n++}`;
    taken.add(id);
    sections.push({ id, label });
    return `<h2 id="${id}">${inner}</h2>`;
  });

  /* One body hand-wrote the wrapper; strip it first so the rule below applies
     exactly once everywhere. These bodies hold no <div> inside a table. */
  const withTables = withIds
    .replace(/<div class="article-table-wrap">\s*([\s\S]*?)<\/div>/g, "$1")
    .replace(/<table>/g, '<div class="article-table-wrap"><table>')
    .replace(/<\/table>/g, "</table></div>");

  return { html: withTables, sections };
}
```

```tsx
/**
 * Which heading the reader is under: a band near the top of the viewport,
 * below the floating bar. When no heading is in the band the last answer
 * stands, because a rail that empties mid-section reads as broken.
 */
function useActiveSection(sections: Section[]): string | null {
  const [active, setActive] = useState<string | null>(null);
  const key = sections.map((s) => s.id).join("|");
  useEffect(() => {
    if (!key) return;
    const els = key.split("|").map((id) => document.getElementById(id)).filter((el): el is HTMLElement => Boolean(el));
    if (!els.length) return;
    const io = new IntersectionObserver(
      (entries) => {
        const hit = entries.find((e) => e.isIntersecting);
        if (hit?.target.id) setActive(hit.target.id);
      },
      { rootMargin: "-104px 0px -68% 0px" },
    );
    els.forEach((el) => io.observe(el));
    return () => io.disconnect();
  }, [key]);
  return active;
}
```
