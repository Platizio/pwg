```ts
import assert from "node:assert/strict";
import test from "node:test";
import { calculateTradeCost, compareGainsTax, formatInr, formatUsd, formatPct } from "../Platizio_Global_Revamp/lib/pricing.ts";
import { RATES } from "../Platizio_Global_Revamp/data/pricingRates.ts";

/* Every figure the pricing page shows is derived from the rate file. These
   pin the three derivations the page states in prose, so a rate change that
   silently changes a sentence fails here first. */

test("the TCS worked example: ₹15,00,000 remitted, ₹5,00,000 taxable, ₹1,00,000 collected", () => {
  const remitted = 1_500_000;
  const taxable = remitted - RATES.tcsThresholdInr;
  assert.equal(formatInr(taxable), "5,00,000");
  assert.equal(formatInr(taxable * RATES.tcsPct), "1,00,000");
});

test("the gains comparator on ₹1,00,000: slab 30,000 against long-term 12,500, saving 17,500", () => {
  const r = compareGainsTax(100_000);
  assert.ok(r);
  assert.equal(formatInr(r.shortTermTax), "30,000");
  assert.equal(formatInr(r.longTermTax), "12,500");
  assert.equal(formatInr(r.difference), "17,500");
});

test("the calculator's floor sentence: below about $345 the $1 minimum sets the brokerage", () => {
  assert.equal(formatUsd(RATES.brokerageMinUsd / RATES.brokeragePct, 0), "345");
  const c = calculateTradeCost(1000, "buy");
  assert.ok(c);
  assert.equal(c.minimumApplied, false);
  assert.equal(formatUsd(c.brokerage), "2.90");
  assert.equal(formatPct(c.effectivePct), "0.35%");
});
```

```tsx
"use client";

import { useState } from "react";
import Link from "next/link";
import { RATES, pct } from "@/Platizio_Global_Revamp/data/pricingRates";
import { calculateTradeCost, formatUsd, formatPct, type TradeSide } from "@/Platizio_Global_Revamp/lib/pricing";

/** Constants, not derived values: the server and the client must agree exactly. */
const DEFAULT_VALUE = "1000";
const DEFAULT_SIDE: TradeSide = "buy";
const PRESETS = ["200", "1000", "5000", "25000"];

/**
 * What a trade actually costs.
 *
 * A schedule listing "SEC fee $0.0000206 per $1 traded" is accurate and
 * unusable; this turns it into a number for the reader's own trade. No fetch
 * and no effect: the whole thing is arithmetic over a constant default, so
 * the server HTML already carries a complete, correct breakdown and only the
 * inputs need JavaScript. It renders inside the page's night panel.
 */
export function Calculator() {
  const [raw, setRaw] = useState(DEFAULT_VALUE);
  const [side, setSide] = useState<TradeSide>(DEFAULT_SIDE);

  const value = Number.parseFloat(raw);
  const cost = calculateTradeCost(value, side);
  const show = (n: number | undefined) => (cost ? `$${formatUsd(n as number)}` : "—");

  return (
    <section className="pr-calc" aria-labelledby="calc-heading">
      <div className="pr-calc-head">
        <p className="eyebrow">Work out your cost</p>
        <h2 id="calc-heading" className="pr-calc-h">What a trade actually costs</h2>
        <p className="pr-calc-intro">
          Enter a trade value and see every charge that applies, including the
          effective rate you end up paying.
        </p>
      </div>

      <div className="pr-calc-grid">
        <div className="pr-calc-inputs">
          <div className="pr-field">
            <label className="pr-calc-label" htmlFor="trade-value">Trade value</label>
            <div className="pr-input-wrap">
              <span className="pr-prefix" aria-hidden="true">$</span>
              <input
                id="trade-value"
                className="pr-input figure"
                type="text"
                inputMode="decimal"
                value={raw}
                onChange={(e) => setRaw(e.target.value)}
                aria-describedby="calc-effective"
              />
            </div>
            <div className="pr-presets">
              {PRESETS.map((p) => (
                <button type="button" key={p} className={`pr-preset figure${raw === p ? " is-active" : ""}`} onClick={() => setRaw(p)}>
                  ${formatUsd(Number(p), 0)}
                </button>
              ))}
            </div>
          </div>

          <fieldset className="pr-field pr-side">
            <legend className="pr-calc-label">Order type</legend>
            <div className="pr-toggle">
              {(["buy", "sell"] as const).map((s) => (
                <label key={s} className={`pr-toggle-option${side === s ? " is-active" : ""}`}>
                  <input type="radio" name="side" value={s} checked={side === s} onChange={() => setSide(s)} />
                  {s === "buy" ? "Buy" : "Sell"}
                </label>
              ))}
            </div>
            <p className="pr-hint">
              {side === "sell" ? "Sell orders also carry the SEC and FINRA fees." : "Buy orders carry no SEC or FINRA fee."}
            </p>
          </fieldset>
        </div>

        <div className="pr-readout">
          <ul className="pr-lines">
            <li>
              <span className="pr-line-label">
                Brokerage
                {cost?.minimumApplied && <span className="pr-tag">minimum applied</span>}
              </span>
              <span className="pr-line-value figure">{show(cost?.brokerage)}</span>
            </li>
            <li>
              <span className="pr-line-label">IGST <em>18% on brokerage</em></span>
              <span className="pr-line-value figure">{show(cost?.igst)}</span>
            </li>
            <li>
              <span className="pr-line-label">IFSCA turnover fee</span>
              <span className="pr-line-value figure">{show(cost?.ifsca)}</span>
            </li>
            <li className={side === "buy" ? "is-muted" : undefined}>
              <span className="pr-line-label">SEC fee <em>sell orders only</em></span>
              <span className="pr-line-value figure">{show(cost?.sec)}</span>
            </li>
          </ul>

          {/* The summary, not the breakdown: announcing every line on each
              keystroke is unusable. Total and effective rate are the outcome. */}
          <div className="pr-total" aria-live="polite">
            <div className="pr-total-row">
              <span>Total cost</span>
              <strong className="figure">{show(cost?.total)}</strong>
            </div>
            <div className="pr-effective" id="calc-effective">
              <span>Effective rate</span>
              <strong className="figure">{cost ? formatPct(cost.effectivePct) : "—"}</strong>
            </div>
          </div>

          {cost?.minimumApplied && (
            <p className="pr-callout">
              Below about ${formatUsd(RATES.brokerageMinUsd / RATES.brokeragePct, 0)}, the
              ${RATES.brokerageMinUsd} minimum sets your brokerage rather than the{" "}
              {pct(RATES.brokeragePct)} rate — which is why the effective rate here is
              higher than {pct(RATES.brokeragePct)}.
            </p>
          )}
        </div>
      </div>

      <p className="pr-note">
        Your bank&apos;s currency conversion is not included above — the rate and margin
        are set by your remitting bank, not by us, and on most trades it is the
        largest single cost. FINRA also charges ${RATES.finraPerShare} per share sold,
        excluded because it depends on share count rather than trade value and is
        typically under $0.05 on a retail order. Figures are an illustration, not tax
        or investment advice. <Link href="/disclaimer">Risk disclosure</Link>
      </p>
    </section>
  );
}
```

```tsx
"use client";

import { useState } from "react";
import Link from "next/link";
import { RATES, pct } from "@/Platizio_Global_Revamp/data/pricingRates";
import { compareGainsTax, formatInr } from "@/Platizio_Global_Revamp/lib/pricing";

const DEFAULT_GAIN = "100000";
const PRESETS = ["50000", "100000", "500000", "1000000"];

/**
 * Short-term against long-term, side by side. The holding period IS the
 * point, so it is a comparison rather than an input. Both assumptions sit
 * inline next to the numbers rather than in a footnote.
 */
export function Gains() {
  const [raw, setRaw] = useState(DEFAULT_GAIN);
  const gain = Number.parseFloat(raw);
  const result = compareGainsTax(gain);
  const show = (n: number | undefined) => (result ? `₹${formatInr(n as number)}` : "—");

  return (
    <div className="pr-gains">
      <div className="pr-field">
        <label className="pr-calc-label" htmlFor="gain-amount">Your capital gain</label>
        <div className="pr-input-wrap">
          <span className="pr-prefix" aria-hidden="true">₹</span>
          <input id="gain-amount" className="pr-input figure" type="text" inputMode="numeric" value={raw} onChange={(e) => setRaw(e.target.value)} />
        </div>
        <div className="pr-presets">
          {PRESETS.map((p) => (
            <button type="button" key={p} className={`pr-preset figure${raw === p ? " is-active" : ""}`} onClick={() => setRaw(p)}>
              ₹{formatInr(Number(p))}
            </button>
          ))}
        </div>
      </div>

      <div className="pr-gains-compare" aria-live="polite">
        <article className="pr-gains-card">
          <h4>Sold within 24 months</h4>
          <p className="pr-gains-rate">Your income slab<em>assumed {pct(RATES.stcgAssumedSlabPct)}, the highest slab</em></p>
          <p className="pr-gains-figure figure">{show(result?.shortTermTax)}</p>
        </article>
        <article className="pr-gains-card is-favoured">
          <h4>Held past 24 months</h4>
          <p className="pr-gains-rate">{pct(RATES.ltcgPct)} flat<em>long-term capital gains</em></p>
          <p className="pr-gains-figure figure">{show(result?.longTermTax)}</p>
        </article>
      </div>

      {result && (
        <p className="pr-gains-delta">
          Holding past {RATES.ltcgThresholdMonths} months saves{" "}
          <strong className="figure">₹{formatInr(result.difference)}</strong> on this gain.
        </p>
      )}

      <p className="pr-note pr-note--paper">
        Your actual slab may be lower than {pct(RATES.stcgAssumedSlabPct)}, and
        surcharge and cess are not included — so the short-term figure understates
        what you would owe. An illustration, not tax advice.{" "}
        <Link href="/disclaimer">Risk disclosure</Link>
      </p>
    </div>
  );
}
```
