```tsx
"use client";

import { useEffect, useState } from "react";
import Link from "next/link";
import { Marquee } from "@/components/marketing/motion/marquee";
import { NEWS, type NewsItem } from "@/Platizio_Global_Revamp/data/mediaNews";
import { sortNews, formatNewsDate } from "@/Platizio_Global_Revamp/lib/mediaSelect";

interface RailItem { kind: string; headline: string; date: string; href: string; external?: boolean }

/** Curated list, newest first: the initial render on both server and client. */
const SEED: RailItem[] = sortNews(NEWS).slice(0, 8).map((n: NewsItem) => ({ ...n }));

function Headline({ item }: { item: RailItem }) {
  const inner = (
    <>
      <span className="md-item-text">{item.headline}</span>
      <span className="md-item-meta">
        <span className="md-item-kind label">{item.kind}</span>
        <time className="md-item-date figure" dateTime={item.date}>{formatNewsDate(item.date)}</time>
      </span>
    </>
  );
  return item.external ? (
    <a className="md-item" href={item.href} target="_blank" rel="noopener noreferrer">{inner}</a>
  ) : (
    <Link className="md-item" href={item.href}>{inner}</Link>
  );
}

/**
 * The news strip. Renders the curated list immediately, identically on
 * server and client, then swaps in live US-market headlines once /api/news
 * answers with at least four. If that fails, or the finite quota runs out,
 * the curated items stay. The strip is never empty.
 */
export function NewsRail() {
  const [items, setItems] = useState<RailItem[]>(SEED);

  useEffect(() => {
    const controller = new AbortController();
    fetch("/api/news", { signal: controller.signal })
      .then((res) => (res.ok ? res.json() : Promise.reject(new Error(String(res.status)))))
      .then((payload: { items?: RailItem[] }) => {
        if (Array.isArray(payload?.items) && payload.items.length >= 4) setItems(payload.items);
      })
      .catch(() => {});
    return () => controller.abort();
  }, []);

  if (!items.length) return null;

  return (
    <section className="md-rail" aria-label="Latest US market news">
      <div className="md-rail-row">
        <p className="md-rail-label">
          <span className="md-rail-dot" aria-hidden="true" />
          Markets
        </p>
        <Marquee label="Latest US market news" spacing="0" duration={70}>
          {items.map((item) => <Headline item={item} key={item.href} />)}
        </Marquee>
      </div>
    </section>
  );
}
```

```tsx
/* maxresdefault (1280x720) and mqdefault (320x180) are both natively 16:9 and
   need no cropping; hqdefault is 4:3 with letterbox bars. maxres is not
   guaranteed for every upload, so the feature falls back to mq once. */
const thumbMax = (id: string) => `https://img.youtube.com/vi/${id}/maxresdefault.jpg`;
const thumbMq = (id: string) => `https://img.youtube.com/vi/${id}/mqdefault.jpg`;

    cookies on page load for every visitor, whether or not they press play. */
export function Watch() {
  const { feature, side } = selectVideos(VIDEOS);
  if (!feature) return null;

  return (
    <section className="md-section band" id="videos" aria-labelledby="mw-watch-h">
```

```tsx
"use client";

import { useState, type FormEvent } from "react";
import Link from "next/link";
import { Reveal } from "@/components/marketing/motion/reveal";

const SUPPORT_EMAIL = "supportglobal@platizio.com";
type Status = "idle" | "sending" | "done" | "error";

/**
 * Newsletter signup, on the page's night panel. Posts to /api/subscribe,
 * which answers 503 until a provider is connected; the form says so plainly
 * and offers an address. It never shows a success it cannot honour.
 */
export function Letter() {
  const [email, setEmail] = useState("");
  const [status, setStatus] = useState<Status>("idle");
  const [message, setMessage] = useState("");

  async function onSubmit(e: FormEvent) {
    e.preventDefault();
    if (status === "sending") return;
    setStatus("sending");
    setMessage("");
    try {
      const res = await fetch("/api/subscribe", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email }) });
      const body = await res.json().catch(() => ({}));
      if (res.ok) {
        setStatus("done");
        setMessage("You are on the list. Look out for the first issue.");
        setEmail("");
      } else {
        setStatus("error");
        setMessage(body?.error || "Could not sign you up just now.");
      }
    } catch {
      setStatus("error");
      setMessage("Could not reach the server. Check your connection and try again.");
    }
  }

  return (
    <section className="md-letter band" aria-labelledby="mw-letter-h">
      <div className="wrap">
        <Reveal className="md-letter-panel night">
          <div className="md-letter-copy">
            <p className="eyebrow">03 — the letter</p>
            <h2 id="mw-letter-h">What we are reading, monthly</h2>
            <p className="md-letter-body">
              One email a month on global markets, tax changes that affect Indian
              investors, and what we have published. No trade calls, no daily noise.
            </p>
          </div>
          <form className="md-form" onSubmit={onSubmit} noValidate>
            <label className="sr-only" htmlFor="newsletter-email">Email address</label>
            <div className="md-controls glass glass--dark" data-glass="">
              <input
                id="newsletter-email"
                className="md-input"
                type="email"
                name="email"
                inputMode="email"
                autoComplete="email"
                placeholder="you@example.com"
                required
                value={email}
                onChange={(e) => { setEmail(e.target.value); if (status !== "idle") setStatus("idle"); }}
                aria-describedby="newsletter-status"
              />
              <button className="btn-gold btn--sm" data-glass="" type="submit" disabled={status === "sending"}>
                {status === "sending" ? "Signing up…" : "Subscribe"}
              </button>
            </div>
            <p className={`md-status is-${status}`} id="newsletter-status" role="status" aria-live="polite">
              {status === "error" && (
                <>
                  <span className="md-status-flag">Not sent — </span>
                  {message}{" "}
                  <a href={`mailto:${SUPPORT_EMAIL}?subject=Newsletter%20signup`}>Email us instead</a>
                </>
              )}
              {status === "done" && (
                <>
                  <span className="md-status-flag">Subscribed — </span>
                  {message}
                </>
              )}
              {status === "idle" && (
                <>
                  We will not share your address. See our{" "}
                  <Link href="/privacy">privacy policy</Link>.
                </>
              )}
            </p>
          </form>
        </Reveal>
      </div>
    </section>
  );
}
```
