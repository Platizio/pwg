export interface LegalMeta { term: string; detail: ReactNode }

export interface LegalDocument {
  seo: { title: string; description: string; canonical: string; crumb: string };
  h1: ReactNode;
  sub: string;
  effective: string;
  meta?: LegalMeta[];
  /** The preamble or intro note, verbatim. */

  tocId: string;
  sections: LegalSection[];
  contact: { title: string; body: ReactNode };

/**
 * One chassis for the three legal documents: the entity list as ruled rows,
 * a sticky contents rail, the numbered sections at the reading measure, the
 * contact card. The text is the content module's and is never touched here.

export function LegalPage({ doc }: { doc: LegalDocument }) {
  return (
    <div className="lg">
      <SEO
        title={doc.seo.title}
        description={doc.seo.description}
        canonical={doc.seo.canonical}
        jsonLd={breadcrumbSchema([["Home", "/"], [doc.seo.crumb, doc.seo.canonical]])}
      />
      <section className="lg-hero">
        <div className="wrap">
          <nav className="lg-crumb" aria-label="Breadcrumb">
            <Link href="/">Home</Link><span className="lg-crumb-sep" aria-hidden="true">/</span><span aria-current="page">{doc.seo.crumb}</span>
          </nav>
          <div className="lg-copy">
            <h1>{doc.h1}</h1>
            <p className="lede">{doc.sub}</p>
            <p className="lg-effective provenance">Effective Date: {doc.effective}</p>
          </div>
        </div>
      </section>

      <section className="lg-doc band">
        <div className="wrap lg-grid">
          <nav className="lg-rail" aria-labelledby={doc.tocId}>
            <h2 id={doc.tocId} className="label">Contents</h2>
            <ol>
              {doc.sections.map((s) => <li key={s.id}><a href={`#${s.id}`}><span className="lg-rail-n figure">{s.n}</span>{s.title}</a></li>)}
            </ol>
          </nav>
          <div className="lg-main">

              <dl className="lg-meta">
                {doc.meta.map(({ term, detail }) => <div className="lg-meta-row" key={term}><dt className="label">{term}</dt><dd>{detail}</dd></div>)}
              </dl>
            )}
            <p className="lg-note">{doc.note}</p>
            {doc.sections.map((s) => (
              <section className="lg-section" id={s.id} key={s.id}>
                <h2><span className="lg-num figure">{s.n}</span>{s.title}</h2>
                {s.body}

              <h3>{doc.contact.title}</h3>

                <span>Also on this site:</span>

- [ ] **Step 1: Write the test that compares old and new text**

```ts
import assert from "node:assert/strict";
import test from "node:test";
import { readFileSync } from "node:fs";
import { fileURLToPath } from "node:url";

/* The three legal documents were moved from JSX views into content modules.
   A legal document that changes by one word in the move is a defect, so this
   reads both files as text, strips everything that is not prose (tags, JSX
   expressions, attributes, imports, comments) and compares what is left. */

const root = fileURLToPath(new URL("../", import.meta.url));

function prose(path: string): string {
  const src = readFileSync(root + path, "utf8");
  return src
    .replace(/\/\*[\s\S]*?\*\//g, " ")
    .replace(/\{\/\*[\s\S]*?\*\/\}/g, " ")
    .replace(/^import .*$/gm, " ")
    .replace(/<[^>]+>/g, " ")
    .replace(/\{[^{}]*\}/g, " ")
    .replace(/&apos;/g, "'")
    .replace(/&amp;/g, "&")
    .replace(/[A-Za-z]+=\"[^\"]*\"/g, " ")
    .replace(/\s+/g, " ")
    .trim();
}

const PAIRS: Array<[string, string]> = [
  ["src/views/TermsAndConditions.tsx", "components/marketing/legal/terms.tsx"],
  ["src/views/PrivacyPolicy.tsx", "components/marketing/legal/privacy.tsx"],
  ["src/views/Disclaimer.tsx", "components/marketing/legal/disclaimer.tsx"],
];

for (const [oldPath, newPath] of PAIRS) {
  test(`${newPath} carries every word of ${oldPath}`, () => {
    const before = prose(oldPath);
    const after = prose(newPath);
    /* The new module also carries the chassis strings (Contents, the crumb),
       so it must contain the old text; the old must be a subsequence of it
       at the sentence level. */
    const sentences = before.split(/(?<=[.!?])\s+/).filter((s) => s.length > 20);
    const missing = sentences.filter((s) => !after.includes(s));
    assert.deepEqual(missing, [], `sentences lost in the move:\n${missing.join("\n")}`);
  });
}
```
