```tsx
/**
 * The section the reader is inside, for the rail. Initialised from static data
 * so the server and first client render agree; the observer runs in an effect.
 */
function useActiveSection(ids: string[]): string {
  const [active, setActive] = useState(ids[0]);
  useEffect(() => {
    const els = ids.map((id) => document.getElementById(id)).filter((el): el is HTMLElement => el !== null);
    if (!els.length || !("IntersectionObserver" in window)) return;
    const io = new IntersectionObserver(
      (entries) => {
        const visible = entries.filter((e) => e.isIntersecting);
        if (!visible.length) return;
        const top = visible.sort((a, b) => a.boundingClientRect.top - b.boundingClientRect.top)[0];
        setActive(top.target.id);
      },
      { rootMargin: "-96px 0px -66% 0px", threshold: 0 },
    );
    els.forEach((el) => io.observe(el));
    return () => io.disconnect();
  }, [ids]);
  return active;
}
```

```js
import { readFileSync, writeFileSync } from "node:fs";

/* Lifts every rule whose selectors are all .assistant-* / .sform-* out of
   css/styles.css into a file the new help sheet starts from, keeping @media
   nesting, and removes them from styles.css. */

const SRC = "css/styles.css";
const OUT = "/private/tmp/claude-501/-Users-aayushsharma-Desktop-PWG-pwg-new/f2a6e964-ea6f-4a8d-9b46-9a86a847d70e/scratchpad/assistant-rules.css";
const MATCH = [/^\.assistant\b/, /^\.assistant-/, /^\.sform\b/, /^\.sform-/, /^@keyframes assistant-/];

function walk(css, keep, take) {
  let i = 0;
  while (i < css.length) {
    const open = css.indexOf("{", i);
    if (open < 0) { keep.push(css.slice(i)); break; }
    const prelude = css.slice(i, open);
    let depth = 1, j = open + 1;
    while (j < css.length && depth) { if (css[j] === "{") depth++; else if (css[j] === "}") depth--; j++; }
    const body = css.slice(open + 1, j - 1);
    const sel = prelude.replace(/\/\*[\s\S]*?\*\//g, "").trim();
    if (/^@(media|supports)/.test(sel)) {
      const innerKeep = [], innerTake = [];
      walk(body, innerKeep, innerTake);
      if (innerTake.join("").trim()) take.push(`${sel} {${innerTake.join("")}}\n`);
      keep.push(prelude + "{" + innerKeep.join("") + "}");
    } else {
      const parts = sel.split(",").map((s) => s.trim()).filter(Boolean);
      const isAssistant = parts.length > 0 && parts.every((s) => MATCH.some((re) => re.test(s)));
      (isAssistant ? take : keep).push(prelude + "{" + body + "}");
    }
    i = j;
  }
}

const src = readFileSync(SRC, "utf8");
const keep = [], take = [];
walk(src, keep, take);
writeFileSync(OUT, take.join("\n"));
writeFileSync(SRC, keep.join(""));
console.log(`took ${take.length} rule blocks into ${OUT}; styles.css now ${keep.join("").split("\n").length} lines`);
```
