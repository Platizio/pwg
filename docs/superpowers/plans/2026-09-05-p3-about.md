### Task 1: The team grid

**Files:**
- Create: `components/marketing/about/team.tsx`

- [ ] **Step 1: Create it**

```tsx
"use client";

import Image from "next/image";
import { TEAM, initials } from "@/Platizio_Global_Revamp/data/team";

/**
 * The whole team, visible at once. A grid needs no autoplay, no interval and
 * no reduced-motion handling, and on a page whose job is "who is behind this",
 * nobody rotates out of view. Static markup: the server HTML is the finished
 * section. Initials sit under each photo; if the image fails it removes
 * itself, revealing them.
 */
export function Team() {
  return (
    <ul className="au-team">
      {TEAM.map((member) => (
        <li className="card au-member" key={member.name}>
          <div className="au-photo">
            <span className="au-initials" aria-hidden="true">{initials(member.name)}</span>
            <Image
              className="au-photo-img"
              src={member.image}
              alt={member.name}
              fill
              sizes="(max-width: 620px) 45vw, (max-width: 1024px) 30vw, 280px"
              draggable={false}
              onError={(e) => { e.currentTarget.style.display = "none"; }}
            />
          </div>
          <p className="au-name">{member.name}</p>
          <p className="au-role">{member.role}</p>
        </li>
      ))}
    </ul>
  );
}
```
