```ts
/**
 * The particle globe's geometry, pure.
 *
 * The canvas renderer in components/marketing/motion/particle-globe.tsx draws
 * whatever this module hands it; nothing here touches the DOM, so the
 * projection, the route and the rotation are all testable with node.
 *
 * Coordinates: a unit sphere with +y north, +z toward the viewer, and +x to
 * the viewer's right. Longitude 0 faces the viewer at phi 0; rotateY turns the
 * globe about its axis so `facingPhi(lon)` brings that longitude to the front.
 */

export type Vec3 = { x: number; y: number; z: number };

export const DEG = Math.PI / 180;

export const NEW_DELHI = { lat: 28.6139, lon: 77.209 } as const;
export const NEW_YORK = { lat: 40.7128, lon: -74.006 } as const;

export function toVec3(lat: number, lon: number): Vec3 {
  const la = lat * DEG;
  const lo = lon * DEG;
  return { x: Math.cos(la) * Math.sin(lo), y: Math.sin(la), z: Math.cos(la) * Math.cos(lo) };
}

/** n points spread evenly over the sphere by the golden angle. Deterministic. */
export function fibonacciSphere(n: number): Vec3[] {
  const out: Vec3[] = [];
  const golden = Math.PI * (3 - Math.sqrt(5));
  for (let i = 0; i < n; i++) {
    const y = 1 - (i / Math.max(1, n - 1)) * 2;
    const r = Math.sqrt(Math.max(0, 1 - y * y));
    const t = golden * i;
    out.push({ x: Math.cos(t) * r, y, z: Math.sin(t) * r });
  }
  return out;
}

export function rotateY(p: Vec3, phi: number): Vec3 {
  const c = Math.cos(phi);
  const s = Math.sin(phi);
  return { x: p.x * c + p.z * s, y: p.y, z: -p.x * s + p.z * c };
}

/** The rotation that brings `lon` to face the viewer. */
export function facingPhi(lon: number): number {
  return -lon * DEG;
}

/** Orthographic projection onto a canvas of the given centre and radius. */
export function project(p: Vec3, radius: number, cx: number, cy: number): { x: number; y: number; depth: number } {
  return { x: cx + p.x * radius, y: cy - p.y * radius, depth: p.z };
}

/** `steps + 1` points along the shorter great-circle arc from a to b. */
export function greatCircle(a: Vec3, b: Vec3, steps: number): Vec3[] {
  const dot = Math.max(-1, Math.min(1, a.x * b.x + a.y * b.y + a.z * b.z));
  const omega = Math.acos(dot);
  const out: Vec3[] = [];
  if (omega < 1e-9) {
    for (let i = 0; i <= steps; i++) out.push({ ...a });
    return out;
  }
  const so = Math.sin(omega);
  for (let i = 0; i <= steps; i++) {
    const t = i / steps;
    const ka = Math.sin((1 - t) * omega) / so;
    const kb = Math.sin(t * omega) / so;
    out.push({ x: ka * a.x + kb * b.x, y: ka * a.y + kb * b.y, z: ka * a.z + kb * b.z });
  }
  return out;
}

/** The share of `values` strictly below `value`, 0..100. Empty set answers 0. */
export function percentile(value: number, values: readonly number[]): number {
  if (values.length === 0) return 0;
  let below = 0;
  for (const v of values) if (v < value) below++;
  return Math.round((below / values.length) * 100);
}
```

```tsx
"use client";

import { useEffect, useRef } from "react";
import {
  NEW_DELHI, NEW_YORK, facingPhi, fibonacciSphere, greatCircle, project, rotateY, toVec3, type Vec3,
} from "@/lib/globe";

/*
 * A sphere of gold points with the route from New Delhi to New York lit.
 *
 * Canvas 2D, one requestAnimationFrame loop, DPR capped at 2, ~1,600 points
 * drawn as 2px squares. The loop pauses when the canvas leaves the viewport
 * or the tab is hidden. Under reduced motion one frame is drawn at rest and
 * nothing moves.
 *
 * The intro turns the globe from Delhi facing the viewer to New York facing
 * the viewer over 1.8s, then drifts. Colours are read once from the tokens on
 * the root element because a canvas cannot resolve var().
 */

const POINTS = 1600;
const INTRO_MS = 1800;
const DRIFT = 0.05; // radians per second at rest
const REST_PHI = facingPhi((NEW_DELHI.lon + NEW_YORK.lon) / 2);

type Props = { className?: string; label?: string };

function readToken(name: string, fallback: string): string {
  if (typeof window === "undefined") return fallback;
  return getComputedStyle(document.documentElement).getPropertyValue(name).trim() || fallback;
}

function easeOut(t: number): number {
  return 1 - Math.pow(1 - t, 3);
}

export function ParticleGlobe({ className, label = "A globe of points with the route from New Delhi to New York" }: Props) {
  const canvasRef = useRef<HTMLCanvasElement>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    const gold = readToken("--foil", "#d9bd8b");
    const goldHi = readToken("--foil-hi", "#f0dcb8");
    const cream = readToken("--night-ink", "#f2ede5");

    const points = fibonacciSphere(POINTS);
    const delhi = toVec3(NEW_DELHI.lat, NEW_DELHI.lon);
    const york = toVec3(NEW_YORK.lat, NEW_YORK.lon);
    const route = greatCircle(delhi, york, 72);

    let dpr = 1;
    let size = 0;
    const measure = () => {
      dpr = Math.min(2, window.devicePixelRatio || 1);
      size = canvas.clientWidth || 600;
      canvas.width = Math.round(size * dpr);
      canvas.height = Math.round(size * dpr);
    };
    measure();

    const phiFrom = facingPhi(NEW_DELHI.lon);
    const phiTo = facingPhi(NEW_YORK.lon);
    let start = 0;
    let phi = reduce ? REST_PHI : phiFrom;
    let tick = 0;
    let frame = 0;
    let visible = true;

    const drawMarker = (p: Vec3, name: string, r: number, cx: number, cy: number) => {
      const q = project(p, r, cx, cy);
      if (q.depth < 0.05) return;
      const a = Math.min(1, (q.depth + 0.2) * 1.4);
      ctx.globalAlpha = a;
      ctx.fillStyle = goldHi;
      ctx.beginPath();
      ctx.arc(q.x, q.y, 3.2 * dpr, 0, Math.PI * 2);
      ctx.fill();
      ctx.strokeStyle = gold;
      ctx.lineWidth = 1 * dpr;
      ctx.beginPath();
      ctx.arc(q.x, q.y, 9 * dpr, 0, Math.PI * 2);
      ctx.stroke();
      ctx.fillStyle = cream;
      ctx.font = `500 ${10 * dpr}px ui-monospace, Menlo, monospace`;
      ctx.textBaseline = "middle";
      ctx.fillText(name, q.x + 14 * dpr, q.y);
      ctx.globalAlpha = 1;
    };

    const draw = () => {
      const w = canvas.width;
      const cx = w / 2;
      const cy = w / 2;
      const r = w * 0.46;
      ctx.clearRect(0, 0, w, w);

      /* the sphere */
      const sz = 2 * dpr;
      for (const p of points) {
        const q = project(rotateY(p, phi), r, cx, cy);
        if (q.depth < -0.2) continue;
        const a = q.depth > 0 ? 0.28 + q.depth * 0.55 : 0.06;
        ctx.globalAlpha = a;
        ctx.fillStyle = gold;
        ctx.fillRect(q.x - sz / 2, q.y - sz / 2, sz, sz);
      }

      /* the route: a dashed arc whose dashes travel from Delhi to New York */
      ctx.lineWidth = 1.2 * dpr;
      ctx.strokeStyle = goldHi;
      for (let i = 0; i < route.length - 1; i++) {
        if (((i + tick) >> 1) % 2 !== 0) continue;
        const a = project(rotateY(route[i], phi), r, cx, cy);
        const b = project(rotateY(route[i + 1], phi), r, cx, cy);
        if (a.depth < 0 || b.depth < 0) continue;
        ctx.globalAlpha = 0.35 + Math.min(a.depth, b.depth) * 0.6;
        ctx.beginPath();
        ctx.moveTo(a.x, a.y);
        ctx.lineTo(b.x, b.y);
        ctx.stroke();
      }
      ctx.globalAlpha = 1;

      drawMarker(rotateY(delhi, phi), "NEW DELHI", r, cx, cy);
      drawMarker(rotateY(york, phi), "NEW YORK", r, cx, cy);
    };

    const loop = (now: number) => {
      frame = 0;
      if (!visible || document.hidden) return;
      if (!start) start = now;
      const elapsed = now - start;
      if (elapsed < INTRO_MS) {
        phi = phiFrom + (phiTo - phiFrom) * easeOut(elapsed / INTRO_MS);
      } else {
        phi = phiTo + ((elapsed - INTRO_MS) / 1000) * DRIFT;
      }
      if (elapsed % 120 < 17) tick++;
      draw();
      frame = requestAnimationFrame(loop);
    };

    const resume = () => {
      if (reduce || frame || !visible || document.hidden) return;
      frame = requestAnimationFrame(loop);
    };

    const ro = new ResizeObserver(() => { measure(); if (reduce) draw(); });
    ro.observe(canvas);

    const io = new IntersectionObserver(([entry]) => {
      visible = entry.isIntersecting;
      if (visible) resume();
    });
    io.observe(canvas);

    const onVisibility = () => resume();
    document.addEventListener("visibilitychange", onVisibility);

    if (reduce) draw();
    else resume();

    return () => {
      if (frame) cancelAnimationFrame(frame);
      ro.disconnect();
      io.disconnect();
      document.removeEventListener("visibilitychange", onVisibility);
    };
  }, []);

  return <canvas ref={canvasRef} className={className} role="img" aria-label={label} />;
}
```

```ts
 * One request for everything the live-route panel shows: the eight popular
 * names, the index distribution and the day's leaders. Fetched in an effect
 * only, so the server and the first client render both see the skeleton.
 */
export function useQuotes(): QuotesState {
  const [state, setState] = useState<QuotesState>({ quotes: null, indexMoves: [], leaders: [], asOf: null, delayed: true });

  useEffect(() => {
    const controller = new AbortController();
    fetch(`/api/quotes?symbols=${encodeURIComponent(SYMBOLS)}&index=1`, { signal: controller.signal })
      .then((res) => {
        if (!res.ok) throw new Error(`quotes responded ${res.status}`);
        return res.json() as Promise<SymbolsResponse>;
      })
      .then((payload) => {
        if (!Array.isArray(payload?.quotes)) throw new Error("quotes payload malformed");
        setState({
          quotes: payload.quotes.filter((q) => typeof q.price === "number"),
          indexMoves: Array.isArray(payload.indexMoves) ? payload.indexMoves : [],
          leaders: Array.isArray(payload.indexLeaders) ? payload.indexLeaders : [],
          asOf: payload.asOf ?? null,
          delayed: payload.delayed !== false,
        });
      })
      .catch((err: Error) => {
        if (err.name === "AbortError") return;
        setState({ quotes: [], indexMoves: [], leaders: [], asOf: null, delayed: true });
      });
    return () => controller.abort();
  }, []);

  return state;
}
```
